def binary_search(A, target, low, high)
    if low > high
        return -1
    mid ← low + floor((high - low) / 2)
    if A[mid] = target
        return mid
    else if target < A[mid]
        return binary_search(A, target, low, mid - 1)
    else
        return binary_search(A, target, mid + 1, high)
